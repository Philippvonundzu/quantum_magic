\begin{frame}{SPT ground states are hypergraph states}
  \begin{columns}[c]
    \column{0.55\textwidth}
    \begin{itemize}
      \item 2D symmetry-protected topological (SPT) phases \emph{must} contain magic.
      \item Levin--Gu / Miller--Miyake ground states:
        \[\ket{\Psi}=U_{CCZ}\,U_{CZ}\,U_{Z}\ket{+}^{\otimes n}\]
        CCZ on every triangle, CZ on edges, $Z$ on vertices.
      \item $f$ is \emph{cubic} $\Rightarrow$ Boolean toolbox applies; also $\Dmax=\Dmin$.
    \end{itemize}
    \column{0.42\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.62]

    \end{tikzpicture}

    \small triangular lattice, CCZ on every triangle\\(shaded: unit cell)

\begin{frame}{How magical are SPT ground states?}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \textbf{Upper bounds} (cut $f$ into unit cells):
    \begin{itemize}
      \item Triangular: $\Dmin=\Dmax<0.56n$
      \item Union Jack: $\Dmin=\Dmax<0.46n$
      \item Expected: $\approx0.28n$ (disjoint CCZs)
    \end{itemize}
    \textbf{But extensive:} measure some qubits in a Pauli basis $\Rightarrow$ $O(n)$ independent CCZ blocks remain $\Rightarrow$ magic $\propto n$.
    \vspace{0.3em}

    \intuition{SPT phases have short-range entanglement and only ``medium'' magic, comparable to a product state ($0.34n$) --- exactly the regime where Pauli MBQC can be universal (Union Jack).}
    \column{0.47\textwidth}
