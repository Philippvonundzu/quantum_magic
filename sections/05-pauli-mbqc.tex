\begin{frame}{Theorem 5: too much magic is useless}
  \begin{alertblock}{Theorem 5}
    Pauli MBQC with a resource state with $\Dmin(\Psi)\ge n-O(\log n)$ gives no superpolynomial speedup over $\BPP$ for problems in $\NP$.
  \end{alertblock}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \centering
    \begin{tikzpicture}[x=0.27cm,y=0.55cm]
      \node[font=\scriptsize\bfseries] at (8,2.0) {medium magic: structured};
      \foreach \i/\h/\c in {0/0.1/stab!60,1/0.05/stab!60,2/0.1/stab!60,3/0.08/stab!60,4/1.6/good,5/0.05/stab!60,6/0.1/stab!60,7/0.06/stab!60,8/0.08/stab!60,9/0.05/stab!60,10/1.3/good,11/0.1/stab!60,12/0.07/stab!60,13/0.05/stab!60,14/0.1/stab!60,15/0.06/stab!60}

      \draw (0,0)--(16,0);
      \begin{scope}[yshift=-2.15cm]

      \foreach \i/\h/\c in {0/0.45/stab!60,1/0.57/stab!60,2/0.42/good,3/0.34/stab!60,4/0.51/stab!60,5/0.55/good,6/0.37/stab!60,7/0.37/stab!60,8/0.55/stab!60,9/0.50/good,10/0.34/stab!60,11/0.42/good,12/0.57/stab!60,13/0.45/stab!60,14/0.33/good,15/0.48/stab!60}

      \draw (0,0)--(16,0);

      \end{scope}
    \end{tikzpicture}
    \column{0.47\textwidth}
    \small
    \begin{enumerate}
      \item Pauli measurements project onto stabilizer codes.
      \item High magic $\Rightarrow$ tiny overlap with all of them: $p(y)\le2^{\,n-k-\Dmin}$ --- outcomes nearly uniform.
      \item To succeed w.p.\ $2/3$, \emph{many} strings must be solutions.
      \item Classical: guess random strings, check with the NP verifier; $\#$tries $\lesssim2^{\,n-\Dmin}=\mathrm{poly}(n)$.
    \end{enumerate}
  \end{columns}
  \vspace{0.2em}
  \intuition{a too-magical state turns the quantum computer into an expensive coin flipper.}
\end{frame}

\begin{frame}{Corollary 6 \& the ``Goldilocks'' picture}
  \begin{block}{Corollary 6}
    The fraction of states (Haar measure) that give a nontrivial advantage in Pauli MBQC is \alert{exponentially small} in $n$. \ {\small(Theorem 3 + Theorem 5)}
  \end{block}

\end{frame}
