\begin{frame}{Where does quantum advantage come from?}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \begin{itemize}
      \item Natural guess: \alert{entanglement}.
      \item But Clifford circuits make maximally entangled states and are still \emph{easy} to simulate classically.
      \item Adding a single kind of extra gate ($T$) makes them universal.
      \item So the resource is something else: \textbf{magic}.
    \end{itemize}
    \vspace{0.3em}
    \intuition{entanglement is the \emph{plumbing}; magic is the \emph{fuel} that flows through it.}

\begin{frame}{Clifford group: gates that map Paulis to Paulis}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \begin{itemize}
      \item Pauli group $\Pgrp$: products of $I,X,Y,Z$.
      \item Clifford group = normaliser of $\Pgrp$:
        \[\Cgrp=\{U:\ UPU^\dagger\in\Pgrp\ \ \forall P\in\Pgrp\}\]
        generated by $H$, $S$, CNOT.
      \item Stabilizer states: $U\ket{0}^{\otimes n}$ with $U\in\Cgrp$\\(Bell, GHZ, cluster, toric code, \dots).
    \end{itemize}
    \intuition{a Clifford gate only \emph{relabels} Paulis. The state's description stays a short list of Pauli ``labels''.}
    \column{0.47\textwidth}
    \centering
    \begin{tikzpicture}[p/.style={circle,draw=stab,fill=stab!15,minimum size=8mm,font=\small},
                        ar/.style={-{Stealth},thick}]
      \node[p] (X) at (0,0) {$X$};
      \node[p] (Z) at (2.6,0) {$Z$};
      \node[p] (Y) at (1.3,-1.9) {$Y$};
      \draw[ar,good!70!black,bend left=25] (X) to node[above,font=\scriptsize]{$H$} (Z);
      \draw[ar,good!70!black,bend left=25] (Z) to node[below,font=\scriptsize]{$H$} (X);
      \draw[ar,good!70!black] (X) to node[left,font=\scriptsize]{$S$} (Y);
      \node[font=\scriptsize,good!70!black] at (1.3,-2.6) {Clifford: Pauli $\mapsto$ Pauli};
      \node[p,draw=magic,fill=magic!15] (X2) at (0,-3.6) {$X$};
      \node[draw=magic,fill=magic!15,rounded corners,font=\small,minimum height=8mm] (S2) at (2.6,-3.6) {$\tfrac{X+Y}{\sqrt2}$};
      \draw[ar,magic] (X2) to node[above,font=\scriptsize]{$T$} (S2);
      \node[font=\scriptsize,magic] at (1.3,-4.3) {$T$: Pauli $\mapsto$ superposition of Paulis};
    \end{tikzpicture}
  \end{columns}
\end{frame}

\begin{frame}{Magic as a distance from the stabilizer polytope}
  \centering
  \begin{tikzpicture}[scale=1.08]
    \stabfig
    \node[anchor=north west,text width=6.3cm,font=\small,align=left] at (3.2,2.1) {%
      \textcolor{stab}{\bfseries Vertices:} the 6 stabilizer states of one qubit.\\[3pt]
      \textcolor{stab}{\bfseries Octahedron = STAB:} their mixtures, the \emph{free}, classically simulable states.\\[5pt]
      \textcolor{good!75!black}{\bfseries Clifford group:} the 24 rotations mapping the octahedron to itself. Cliffords only shuffle vertices $\Rightarrow$ never leave STAB.\\[5pt]
      \textcolor{magic}{\bfseries $T$ gate:} $\pi/4$ rotation, pushes $\ket{+}$ \emph{outside}. Dotted lines: the ``distance'' magic measures quantify.\\[5pt]
      \textbf{$n$ qubits:} polytope in $4^n\!-\!1$ dimensions with $2^{\Theta(n^2)}$ vertices.};
  \end{tikzpicture}
\end{frame}
