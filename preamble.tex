% the single-qubit stabilizer octahedron
\newcommand{\stabfig}{%
  % ---- single-qubit stabilizer octahedron inside the Bloch sphere ----
  % orthographic view (azimuth 35deg, elevation 20deg), sphere radius 2
  \coordinate (O)  at (0,0);
  \coordinate (zp) at (0,1.879);     \coordinate (zm) at (0,-1.879);
  \coordinate (xp) at (-1.147,-0.56); \coordinate (xm) at (1.147,0.56);
  \coordinate (yp) at (1.638,-0.392); \coordinate (ym) at (-1.638,0.392);
  % sphere
  \shade[inner color=white,outer color=gray!28] (O) circle (2);
  \draw[gray!70] (O) circle (2);
  % equator: back half dashed, front half solid
  \draw[gray!60,dashed,domain=125:305,samples=60,smooth]
    plot ({2*(-0.574*cos(\x)+0.819*sin(\x))},{2*(-0.28*cos(\x)-0.196*sin(\x))});
  \draw[gray!70,domain=-55:125,samples=60,smooth]
    plot ({2*(-0.574*cos(\x)+0.819*sin(\x))},{2*(-0.28*cos(\x)-0.196*sin(\x))});
  % hidden octahedron edges
  \draw[stab!70,dashed] (xm)--(ym) (xm)--(zm) (ym)--(zm);
  % front faces
  \fill[stab!22,opacity=0.85] (zp)--(xp)--(yp)--cycle;
  \fill[stab!14,opacity=0.85] (zm)--(xp)--(yp)--cycle;
  \fill[stab!30,opacity=0.85] (zp)--(xp)--(ym)--cycle;
  \fill[stab!30,opacity=0.85] (zp)--(yp)--(xm)--cycle;
  % visible edges
  \draw[stab,thick] (zp)--(xp)--(zm)  (zp)--(yp)--(zm) (xp)--(yp)
                    (zp)--(ym)--(xp) (zp)--(xm)--(yp) (ym)--(zm) ;
  % vertices = the 6 single-qubit stabilizer states
  \foreach \p in {zp,zm,xp,xm,yp,ym} \fill[stab] (\p) circle (2.3pt);
  \node[above=2pt,font=\small] at (zp) {$\ket{0}$};
  \node[below=2pt,font=\small] at (zm) {$\ket{1}$};
  \node[below left=0pt,font=\small] at (xp) {$\ket{+}$};
  \node[above right=0pt,font=\small] at (xm) {$\ket{-}$};
  \node[right=2pt,font=\small] at (yp) {$\ket{+i}$};
  \node[left=2pt,font=\small] at (ym) {$\ket{-i}$};
  % H gate: |0> -> |+>, vertex -> vertex (arc outside the sphere)
  \draw[good!75!black,very thick,-{Stealth},domain=6:84,samples=40,smooth]
    plot ({2.35*(-0.574*sin(\x))},{2.35*(0.9395*cos(\x)-0.28*sin(\x))});
  \node[good!75!black,font=\scriptsize\bfseries,align=right,anchor=east] at (-1.75,1.45) {$H$ (Clifford)\\[-1pt]vertex$\to$vertex};
  % T gate: |+> -> T|+>, leaves the polytope
  \coordinate (Tst) at (0.347,-0.674);
  \draw[magic,very thick,-{Stealth},domain=4:43,samples=30,smooth]
    plot ({2*(-0.574*cos(\x)+0.819*sin(\x))},{2*(-0.28*cos(\x)-0.196*sin(\x))});
  \fill[magic] (Tst) circle (2.6pt);
  \node[magic,font=\small,anchor=north] at ($(Tst)+(0.15,-0.06)$) {$T\ket{+}$};
  % distance from nearest edge of the polytope
  \coordinate (Emid) at (0.246,-0.476);
  \draw[magic,densely dotted,thick] (Tst)--(Emid);
  % T-type magic state above the (+,+,+) face
  \coordinate (F) at (0.164,0.309); \coordinate (M) at (0.284,0.535);
  \fill[magic] (M) circle (2.3pt);
  \draw[magic,densely dotted,thick] (F)--(M);
  \node[magic,font=\scriptsize,anchor=south west] at ($(M)+(0.02,0.02)$) {$\ket{F}$};

}
